安装数据传输软件
sudo apt install cifs-utils

1、创建本地目录，用于关在本地GPU训练服务器样本传输、下载数据至数据共享服务空间。
sudo mkdir /mnt/share

2、 挂载数据共享服务空间到本地目录
sudo mount -t cifs \
-o username=smb_share,password=smb123,iocharset=utf8,uid=1000,gid=1000,file_mode=0777,dir_mode=0777 \
//172.16.50.250/share \
/mnt/share

    sudo mount -t cifs \
        -o username=smb_share,password=smb123,iocharset=utf8,uid=1000,gid=1000 \
        //172.16.50.250/share \
        /mnt/share

3、查看挂载是否成功
df -h | grep /mnt/share

卸载数据共享服务空间
sudo umount /mnt/share

4、传输本都地GPU训练服务器样本至数据共享服务空间
rsync -avh --progress \
robot-server@172.16.50.229:/home/robot-server/data/AITotal_SegmentDatabase/test.my \
/mnt/share/算法领域/数据集/AI数据集/AISampleDataset/AITotal_SegmentDatabase/
